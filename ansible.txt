cat > ${WORK_FOLDER}/roles/mysql/tasks/install_mysql.yml<<EOF
---
- name: "Installing Percona MySQL Server Version {{ mysql_version }}"
  hosts: dbservers
  vars_files:
        - "${VARS_FOLDER}/${VARS_FILE}"
  become: true
  gather_facts: yes
  tasks:
        - name: "Creating Necessary Directories"
          become: true
          file:
                path: "{{ item }}"
                state: directory
                owner: mysql
                group: mysql
                recurse: yes
          loop:
        - "{{ mysql_directories_location }}"
        - "{{ mysql_datadir_location }}"
        - "{{ mysql_logdir_location }}"
        - "{{ mysql_tempdir_location }}"
        - "{{ mysql_binlogdir_location }}"
        - "{{ mysql_relaylogdir_location }}"
        - "{{ mysql_conf_location }}"
        - "{{ mysql_ssl_location }}"

        - name: "Ensure below file exists"
          become: true
          file:
                path: "{{ item }}"
                state: touch
                owner: mysql
                group: mysql
          loop:
                - "{{ mysql_conf_location }}/my.cnf"
                - "/etc/systemd/system/dbteam-{{ mysql_app }}-{{ mysql_env }}-mysql.service"

        - name: "Copying my.cnf template"
          become: true
          template:
                src: "${WORK_FOLDER}/roles/mysql/templates/my.cnf.j2"
                dest: "{{ mysql_conf_location }}/my.cnf"
                owner: mysql
                group: mysql

        - name: "Copying mysql.service template"
          become: true
          template:
                src: "${WORK_FOLDER}/roles/mysql/templates/mysql-service.j2"
                dest: "/etc/systemd/system/dbteam-{{ mysql_app }}-{{ mysql_env }}-mysql.service"

        - name: "Reload daemon"
          become: true
          systemd:
                daemon_reload: yes

        - name: "enable mysql service"
          become: true
          systemd:
                name: "dbteam-{{ mysql_app }}-{{ mysql_env }}-mysql.service"
                enabled: yes

        - name: "Disabling Group Replication Parameters"
      become: true
      shell: |
        config_file="{{ mysql_conf_location }}/my.cnf"
                config_file_bkp="\${config_file}.\$(date +%s)"
                cp "\${config_file}" "\${config_file_bkp}"
                sed -i 's/^\s*\(group_.*\)=/# \1=/' "\${config_file}"


        - name: "Initializing the data directory"
          become: true
          shell: |
                cd {{ mysql_basedir_location }}
                bin/mysqld --defaults-file={{ mysql_conf_location }}/my.cnf --initialize --user=mysql

        - name: "Copying ssl certs into {{ mysql_conf_location }}"
          become: true
          copy:
                src: "{{ item.src }}"
                dest: "{{ item.dest }}"
                owner: mysql
                group: mysql
          loop:
                - { src: "${WORK_FOLDER}/roles/mysql/templates/server-key.pem", dest: "{{ mysql_ssl_location }}/server-key.pem" }
                - { src: "${WORK_FOLDER}/roles/mysql/templates/server-cert.pem", dest: "{{ mysql_ssl_location }}/server-cert.pem" }
                - { src: "${WORK_FOLDER}/roles/mysql/templates/public_key.pem", dest: "{{ mysql_ssl_location }}/public_key.pem" }
                - { src: "${WORK_FOLDER}/roles/mysql/templates/private_key.pem", dest: "{{ mysql_ssl_location }}/private_key.pem" }
                - { src: "${WORK_FOLDER}/roles/mysql/templates/client-key.pem", dest: "{{ mysql_ssl_location }}/client-key.pem" }
                - { src: "${WORK_FOLDER}/roles/mysql/templates/client-cert.pem", dest: "{{ mysql_ssl_location }}/client-cert.pem" }
                - { src: "${WORK_FOLDER}/roles/mysql/templates/ca.pem", dest: "{{ mysql_ssl_location }}/ca.pem" }
                - { src: "${WORK_FOLDER}/roles/mysql/templates/ca-key.pem", dest: "{{ mysql_ssl_location }}/ca-key.pem" }

        - name: "Starting MySQL Service"
          become: true
          systemd:
                name: "dbteam-{{ mysql_app }}-{{ mysql_env }}-mysql.service"
                state: started

        - name: "Wait for MySQL port to be open"
          wait_for:
                port: "{{ mysql_port }}"
                state: started
                timeout: 300

        - name: "Getting the Temporary password"
          become: true
          shell: |
                grep 'temporary password' {{ mysql_log_error }} | awk '{print \$NF}' | tail -1
          register: mysql_temp_password

        - name: "Printing the temporary password"
          debug:
                var: mysql_temp_password.stdout

          become: true
          shell: |
                {{ mysql_basedir_location }}/bin/mysql -u{{ mysql_super_user }} -p'{{ mysql_temp_password.stdout }}' --socket={{ mysql_socket }} --connect-expired-passw
ord --execute="ALTER USER '{{ mysql_super_user }}'@'localhost' IDENTIFIED BY '{{ mysql_super_password }}';FLUSH PRIVILEGES;"

cat > ${WORK_FOLDER}/roles/mysql/tasks/configure_cluster.yml<<EOF
---
- name: "Configure cluster"
  hosts: dbservers
  vars_files:
        - "${VARS_FOLDER}/${VARS_FILE}"
  become: true
  gather_facts: yes
  any_errors_fatal: true
  tasks:
        - name: "Checking if the nodes are part of the cluster"
          become: true
          shell: |
                {{ mysql_shell_basedir_location }}/bin/mysqlsh {{ mysql_cluster_admin_user }}@{{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ m
ysql_port }} --password={{ mysql_cluster_admin_user_password }} --js -e "print(dba.getCluster().status())"
          register: clusterstatus
      failed_when: false

        - name: "Printing Cluster Status for {{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_port }}"
      debug:
        var: clusterstatus.rc

    - name: "Exiting if the node is already configured as a cluster"
      fail:
        msg: "Node {{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_port }} is already part of InnoDB Cluster. Exiting..."
      when: clusterstatus.rc == 0

        - name: "Proceeding with the Cluster Initialization and Setup"
      debug:
        msg: "Node {{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_port }} is Not part of InnoDB Cluster. Proceeding..."
      when: clusterstatus.rc != 0

        - name: "Checking the Instance Configuration"
      become: true
      shell: |
        {{ mysql_shell_basedir_location }}/bin/mysqlsh {{ mysql_cluster_admin_user }}@{{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_por
t }} --password={{ mysql_cluster_admin_user_password }} --js -e "dba.checkInstanceConfiguration()"
      register: instanceconfiguration_status
      failed_when: false
          when: clusterstatus.rc != 0

        - name: "Printing Instance configuration for {{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_port }}"
      debug:
        var: instanceconfiguration_status

    - name: "Exiting if the node is not configured properly to setup as a member of the cluster"
      fail:
        msg: "Node {{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_port }} Is having issues. Need to update the config accordingly. Exiti
ng..."
      when:
        - clusterstatus.rc != 0
        - instanceconfiguration_status.rc != 0 or ('invalid' in instanceconfiguration_status.stderr | lower)

    - name: "Creating the cluster on the primary Node"
      become: true
      shell: |
        {{ mysql_shell_basedir_location }}/bin/mysqlsh {{ mysql_cluster_admin_user }}@{{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_por
t }} --password={{ mysql_cluster_admin_user_password }} --js -e "dba.createCluster('{{ mysql_app }}_{{ mysql_env }}_{{ mysql_dc }}_cluster')"
      when: "hostvars[inventory_hostname]['ansible_default_ipv4']['address'] in groups['primary']"

    - name: "Checking the cluster status on the Primary Node"
      become: true
      shell: |
        {{ mysql_shell_basedir_location }}/bin/mysqlsh {{ mysql_cluster_admin_user }}@{{ hostvars[inventory_hostname]['ansible_default_ipv4']['address'] }}:{{ mysql_por
t }} --password={{ mysql_cluster_admin_user_password }} --js -e "print(dba.getCluster().status())"
      register: primarynodeclusterstatus
      failed_when: false
      when: "hostvars[inventory_hostname]['ansible_default_ipv4']['address'] in groups['primary']"

        - name: "Printing Cluster Status on the Primary Node"
      debug:
        var: primarynodeclusterstatus.rc
      when: "hostvars[inventory_hostname]['ansible_default_ipv4']['address'] in groups['primary']"

function log_messages()
{
  message=$1
  info=$2
  echo "[`date +%Y\-%m\-%d\ %H\:%M\:%S`] [$info]: $message"
}


function send_email()
{
  message=$1
  status=$2
  subject=$3

  echo "message --> $message"
  echo "status --> $status"
  echo "subject --> $subject"

  if [[ $status = "ERROR" ]]; then
  echo "$message. Please check script error log.Script is located at [$HOSTNAME:$SCRIPT]" | mailx -r "DatabaseBackup-Monitor" -s "$subject" "$EMAIL_RECIPIENTS"
  exit 0
  else
  echo "$message. Script is located at [$HOSTNAME:$SCRIPT]" | mailx -r "DatabaseBackup-Monitor" -s "$subject" "$EMAIL_RECIPIENTS"
  fi
}

function check_mysql()
{
  mysql_alive=`$MYSQL_BASEDIR/mysqladmin --defaults-file=$DEFAULTS_FILE --connect-timeout=30 ping`
  if [[ $? -ne 0 ]]; then
  log_messages "Status of MySQL is [ $mysql_alive ]" "ERROR"
  log_messages "Cannot get status of MySQL" "ERROR"
  message="Cannot get status of MySQL based on defaults file [$DEFAULTS_FILE]"
  info="ERROR"
  subject="Cannot get status of MySQL"
  send_email "$message" "$info" "$subject"
  exit 0
  else
  log_messages "Status of MySQL is [ $mysql_alive ]" "INFO"
  if [[ $mysql_alive == "mysqld is alive" ]]; then
    if [[ $IS_REPLICA == 1 ]]; then
    log_messages "Checking Replication Status" "INFO"
    replicationinfo=($($MYSQL_BASEDIR/mysql --defaults-file=$DEFAULTS_FILE --execute="SHOW REPLICA STATUS" -nE | grep -E 'Replica_IO_Running:|Replica_SQL_Running:|Secon
ds_Behind_Source:' | awk '{ gsub(/ /,""); print }'))
    if [[ ${replicationinfo[0]} == "Replica_IO_Running:Yes" && ${replicationinfo[1]} == "Replica_SQL_Running:Yes" ]]; then
      log_messages "Replication Status is [${replicationinfo[0]} | ${replicationinfo[1]} | ${replicationinfo[2]}]" "INFO"
    else
      log_messages "Replication Status is [${replicationinfo[0]} | ${replicationinfo[1]}  | ${replicationinfo[2]} ]" "ERROR"
      message="Replication Status is [${replicationinfo[0]} | ${replicationinfo[1]}  | ${replicationinfo[2]} ]"
      info="ERROR"
      subject="Replication Issue"
      send_email "$message" "$info" "$subject"
    fi
    elif [[ $IS_REPLICA == 0 ]]; then
    log_messages "Skipping Replication Check" "INFO"
    else
    log_messages "Unrecognized Param" "ERROR"
    fi
  else
    log_messages "Status of MySQL is [ $mysql_alive ]" "ERROR"
    log_messages "Sending Email" "ERROR"
    message="MySQL is Down. Cannot connect to MySQL"
    info="ERROR"
    subject="MySQL is Down"
    send_email "$message" "$info" "$subject"

  fi
  fi

}

function physical_backup()
{
  log_messages "Checking if full Physical Backup exists" "INFO"
  xtrabackup_info_cnt=`aws s3 ls $S3_LOCATION/FULL/xtrabackup_info | wc -l`
  if [[ $xtrabackup_info_cnt -gt 0 ]]; then
  log_messages "Full Physical Backup exists.Proceeding with incremental" "INFO"
  log_messages "Checking if any incremental job running" "INFO"
  if [[ -f "/tmp/$ENV-$BACKUP_DIR-incr.txt" ]]; then
    log_messages "There is already an incremental backup job running." "WARN"
    exit 0
  else
    touch /tmp/$ENV-$BACKUP_DIR-incr.txt
    mkdir -p /tmp/$ENV/LSN
    aws s3 cp $S3_LOCATION/LSN/ /tmp/$ENV/LSN/ --recursive
    incr_dir_cnt=`aws s3 ls $S3_LOCATION/INCR | wc -l`
    if [[ -z $incr_dir_cnt ]]; then
    incr_dir_cnt=0
    fi
    NUMBER=$(($incr_dir_cnt + 1))
    $XTRABACKUP_BASEDIR/xtrabackup --defaults-file=$DEFAULTS_FILE --backup --history --safe-slave-backup --slave-info --stream=xbstream --extra-lsndir=/tmp/$ENV/LSN --t
arget-dir=/tmp/$ENV/LSN --incremental-basedir=/tmp/$ENV/LSN --extra-lsndir=/tmp/$ENV/LSN | $XTRABACKUP_BASEDIR/xbcloud put $S3_LOCATION/INCR$NUMBER --s3-access-key=${AW
S_ACCESS_KEY_ID} --s3-secret-key=${AWS_SECRET_ACCESS_KEY}  --parallel=$PARALLEL_UPLOADS inc_backup
    if [[ $? -ne 0 ]]; then
    log_messages "Error While Taking Incremental backup for env [$ENV] into location [$S3_LOCATION/INCR]" "ERROR"
    log_messages "Sending Email" "ERROR"
    message="Issue while Taking Incremental backup"
    info="ERROR"
    subject="Alter | DB Backup | $ENV - Issue While Taking Incremental backup"
    send_email "$message" "$info" "$subject"
    else
    log_messages "Done Taking Incremental Backup for $ENV into [$S3_LOCATION/INCR]" "INFO"
    echo "Incremental backup [INCR$NUMBER] can be used to restore till --> " `cat /tmp/$ENV/LSN/xtrabackup_info | grep end_time | awk -F"=" '{print $2}'` >> /tmp/$ENV/L
SN/Restore_till.txt
    aws s3 cp /tmp/$ENV/LSN $S3_LOCATION/LSN --recursive
    rm -rf /tmp/$ENV-$BACKUP_DIR-incr.txt
    fi
  fi
  else
  log_messages "Full Backup Doesn't exists.Proceeding with Full Physical Backup" "INFO"

    touch /tmp/$ENV-$BACKUP_DIR-full.txt

arget-dir=/tmp/$ENV  | $XTRABACKUP_BASEDIR/xbcloud put $S3_LOCATION/FULL --s3-access-key=${AWS_ACCESS_KEY_ID} --s3-secret-key=${AWS_SECRET_ACCESS_KEY}  --parallel=$PARA
LLEL_UPLOADS full_backup

  fi
